require 'sinatra' # <1>

class Greeter < Base
  attr_reader :name

  def initialize(name = "World")
    @name = name
  end

  def greet
    puts "Hello, #{@name}!" if @name =~ /\w+/
    %w(a b c).map(&:upcase)
  end
end <2>
